    run = list_of_points[i + 1]['x'] - list_of_points[i]['x']
    euc_distance = math.sqrt(rise*rise + run*run)
    distance_ratio = euc_distance / test_grid.unit
    temp_points = [None] * (math.floor(distance_ratio) + 2)
    temp_points[0] = list_of_points[i] # pierwszym punktem pośrednim jest oryginalny punkt przy którym aktualnie jesteśmy
