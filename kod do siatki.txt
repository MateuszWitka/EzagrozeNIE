# to jest klasa kwadrat, to będzie węzeł
# y i x to koordynaty granic województwa - szerokość i długość, najpierw południe, potem północ, najpierw zachód, potem wschód, dalsze kierunki to sąsiednie kwadraty:

class Square:

    def __init__(self, data = None, y = [None, None], x = [None, None], down = None, up = None, left = None, right = None):
        self.data = data
        self.y = y
        self.x = x
        self.down = down
        self.up = up
        self.left = left
        self.right = right


# jedna sekunda zapisana dziesiętne to 1/3600 = 0.0002777777777777778

# to jest klasa siatka:

class Grid:

    def __init__(self, data = None, unit = None, y = [None, None], x = [None, None], down = None, up = None, left = None, right = None):
        self.data = data
# data to będzie dataframe ze zdarzeniami z całego województwa
        self.unit = unit
# unit to granulacja danych, najmniejsza jednostka tzn. bok jednego kwadratu dziesiętnie
        self.y = y
        self.x = x
        self.up = down
        self.down = up
        self.left = left
        self.right = right
        self.root = Square(y = [(self.y[0] + self.y[1])/2 - self.unit/2, (self.y[0] + self.y[1])/2 + self.unit/2], x = [(self.x[0] + self.x[1])/2 - self.unit/2, (self.x[0] + self.x[1])/2 + self.unit/2])
# korzeń to jest środek województwa (średnia z wymiaru) plus minus pół sekundy czyli środkowy kwadrat jedna sekunda kwadratowa powierzchni
