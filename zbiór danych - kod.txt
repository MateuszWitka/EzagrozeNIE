# Istotne rzeczy w pliku xml z wypadkami:
# - <ID> to id zdarzenia
# - <GPS_X_GUS> i <GPS_Y_GUS> to są punkty w środku miejscowości (albo gminy?) oznaczające tylko tą miejscowość a nie miejsce zdarzenia, one są takie same dla różnych zdarzeń w tej samej okolicy, to jest w postaci dziesiętnej, to jest niepotrzebne
# - <WSP_GPS_X> i <WSP_GPS_Y> to są współrzędne miejsca zdarzenia, gwiazdka to stopnie, to jest w postaci tej takiej minutowej-sekundowej tylko przed ostatnią cyfrą chyba musi być kropka, pokrywa się z adresem i numerem domu
# - <ULICA_ADRES> i <NUMER_DOMU> to może być adres miejsca zdarzenia
# - <DATA_ZDARZ> i <GODZINA_ZDARZ> to czas zdarzenia


import xml.etree.ElementTree as ET

with open(r'D:\semestr zimowy 25\systemy informatyczne\ćw 4\dane\sewik\xml\2024\SEWIK_EXP_N_XML_02_WOJ__LUBUSKIE_2024.xml', encoding = 'utf8') as f:
    file = f.read()

from bs4 import BeautifulSoup

soup = BeautifulSoup(file, 'xml')

soup_text = soup.prettify()

print(soup_text[0:2000])

root = ET.fromstring(soup_text)

# trzeba przekonwertować ten format minutowy na dziesiętny żeby można było policzyć dystans
# ale najpierw trzeba zrobić tabelę z id zdarzenia, czasem i miejscem

# problem, nie wszystkie zdarzenia mają koordynaty gps i root.iter() pominie te nieobecne i wektor będzie za krótki (na końcu będzie miał puste):
len(root.findall('./ZDARZENIA/ZDARZENIE/WSP_GPS_X'))
len(root.findall('./ZDARZENIA/ZDARZENIE')) # różne długości, 11152 vs 11150

# liczba zdarzeń, tyle będzie wierszy w dataframe'ie:
number_of_rows = len(root.findall('./ZDARZENIA/ZDARZENIE'))


# najpierw długość geograficzna...

x_vector = [None] * number_of_rows
n = 0
for i in root.iter('ZDARZENIE'):
    try:
        x_vector[n] = i.find('WSP_GPS_X').text
    except AttributeError:
        pass
    finally:
        n = n + 1
# po zainicjowaniu wektora wstawiamy do niego koordynaty kolejnych zdarzeń a jeśli nie ma koordynatów to zostaje puste pole
# AttributeError to jest to że brak danych nie ma atrybutu .text
# btw na przyszłość, jeśli byłoby Exception zamiast AttributeError to to podobno robi to że nie zawiera np. keyboard interrupt a samo except to po prostu każdy error
# poprzednio miałem wpisane 10528 w długości wektora, nie wiem czemu, zdarzeń jest 11152

import re

# wyciąganie stopni:
x_degrees = [None] * number_of_rows
for i in range(len(x_vector)):
    try:
        if ('.' in x_vector[i]):
           x_degrees[i] = float(re.findall('\n    (.*)\n', x_vector[i])[0])
        else:
            x_degrees[i] = int(re.findall('\n    (.*)\*', x_vector[i])[0])
    except TypeError:
        pass
# type error jest wtedy jak próbujemy indeksować brak danych
# ten if z kropką jest dlatego że niektóre zdarzenia już mają wpisaną lokalizację w postaci dziesiętnej więc nie trzeba (a nawet nie da się) wyciągać osobno stopni, minut i sekund

# minut:
x_minutes = [None] * number_of_rows
for i in range(len(x_vector)):
    try:
        if ('.' in x_vector[i]):
           x_minutes[i] = 0
        else:
            x_minutes[i] = int(re.findall('\*(.*)\'', x_vector[i])[0])
    except TypeError:
        pass
# jeśli mamy oryginalnie zapis dziesiętny (z kropką) to robimy zero żeby potem dodało się zero na sam koniec

# i sekund:
x_seconds = [None] * number_of_rows
for i in range(len(x_vector)):
    try:
        if ('.' in x_vector[i]):
           x_seconds[i] = 0
        else:
            x_seconds[i] = float(re.findall('\'(.*)\n   ', x_vector[i])[0])/10
    except TypeError:
        pass
# dzielenie jest po to żeby ustawić kropkę (przecinek) przed ostatnią cyfrą

# i łączenie w jedne koordynaty dziesiętne:
x_coordinates_decimal = [None] * number_of_rows
for i in range(len(x_coordinates_decimal)):
    try:
        x_coordinates_decimal[i] = x_degrees[i] + (x_seconds[i] + x_minutes[i] * 60) / 3600
    except TypeError:
        pass

from datetime import datetime

# wektor z nieprzetworzoną datą:

    try:
        date_vector[n] = i.find('DATA_ZDARZ').text
    except AttributeError:
        pass

        n = n + 1

# wektor z nieprzetworzoną godziną:
time_vector = [None] * number_of_rows
n = 0
for i in root.iter('ZDARZENIE'):

        time_vector[n] = i.find('GODZINA_ZDARZ').text
    except AttributeError:

# łączenie daty i godziny we wspólny wektor w czasie uniksowym:
unix_time_vector = [None] * number_of_rows
for i in range(len(unix_time_vector)):
    unix_time_vector[i] = datetime.strptime(re.findall('\n    (.*)\n', date_vector[i])[0] + ',' + re.findall('\n    (.*)\n', time_vector[i])[0], '%Y-%m-%d,%H:%M:%S').timestamp()
# usuwamy niepotrzebne spacje i entery regeksami, konwertujemy string na zmienną datetime a potem na timestamp uniksowy
